using System.Diagnostics;
using System.Reflection;
using System.Text.Json;
using FTPlus.AgentGraphicDesign.Core.Pixels;

// Test-only separate process. Reflection observes admission; it does not mutate runtime state.
var options = new JsonSerializerOptions { PropertyNamingPolicy = JsonNamingPolicy.CamelCase };
var service = new NativePixelService();
var slot = typeof(NativePixelService).Assembly.GetType("FTPlus.AgentGraphicDesign.Core.Pixels.NativeOperationScope", true)!
    .GetField("_hostSlot", BindingFlags.Static | BindingFlags.NonPublic)!;
var work = Task.Run(async () => args[0] == "create"
    ? (object)await service.ImportAsync(new(args[1], args[2], "Imported", "Source"))
    : await service.ImportLayerAsync(new(args[1], args[2], JsonSerializer.Deserialize<PixelSnapshot>(File.ReadAllText(args[3]), options)!, "Appended")));
var timer = Stopwatch.StartNew();
while ((int)slot.GetValue(null)! != 1 && !work.IsCompleted && timer.Elapsed < TimeSpan.FromSeconds(5))
    Thread.Sleep(1);
if ((int)slot.GetValue(null)! != 1) throw new Exception("Worker never reached native admission");
Console.WriteLine("ADMITTED");
Console.Out.Flush();
try
{
    var result = await work;
    Console.WriteLine(JsonSerializer.Serialize(new { ok = true, data = result }, options));
}
catch (NativePixelException error)
{
    Console.WriteLine(JsonSerializer.Serialize(new { ok = false, error = new { code = error.Code, message = error.Message } }, options));
    Environment.ExitCode = 1;
}
